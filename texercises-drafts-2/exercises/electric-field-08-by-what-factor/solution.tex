\begin{abcliste}
\abc Point charge: $E = k\,\frac{Q}{r^2}$. The distance counts squared:
\begin{align}
 \frac{E'}{E} &= \frac{2}{3^2} = \result{\frac{2}{9}}
\end{align}

\abc Long wire with the charge $\lambda$ per length: $E = \frac{\lambda}{2\pi\varepsilon_0 r}$. Here the distance counts only once:
\begin{align}
 \frac{E'}{E} &= \frac{3}{1/2} = \result{6}
\end{align}

\abc Large plate with the charge $\sigma$ per area: $E = \frac{\sigma}{2\varepsilon_0}$ does not depend on the distance at all:
\begin{align}
 \frac{E'}{E} &= \result{\frac{1}{2}}
\end{align}

\abc Plate capacitor: $E = \frac{Q}{\varepsilon_0 A}$ depends on the charge per area, not on the distance between the plates:
\begin{align}
 \frac{E'}{E} &= \frac{2}{1/2} = \result{4}
\end{align}
\end{abcliste}
