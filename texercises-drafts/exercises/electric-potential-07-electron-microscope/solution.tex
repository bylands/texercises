\begin{abcliste}
\abc Through the voltage $U$, an electron gains
\begin{align}
 E_\mathrm{kin} &= \EkF = e\times\UO = \result{\EkP{3}{3}}
\end{align}

\abc The rest energy of the electron is
\begin{align}
 E_0 &= \EzF = \Ez \approx \EzP{3}{3}
\end{align}
The kinetic energy is $E_\mathrm{kin}/E_0 = \EkP{3}{3}/\EzP{3}{3} \approx \pctP{0}{2}$ of it, not small: \emph{relativity is needed}.

\abc The total energy is $E = E_0 + E_\mathrm{kin} = \gamma E_0$:
\begin{align}
 \gamma &= \gmF = \gm \\
 \frac{v}{c} &= \btF = \bt \approx \result{\btP{0}{2}}
\end{align}
The classical formula would give $v/c = \bcF = \bcP{0}{2}$, much too fast.
\end{abcliste}
