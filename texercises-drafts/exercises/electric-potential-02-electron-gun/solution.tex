\begin{abcliste}
\abc The electron gains the energy $e U$ as kinetic energy, $\frac{1}{2} m_e v^2 = e U$:
\begin{align}
 v &= \velF \\
   &= \sqrt{\frac{2\times\nce\times\U}{\ncme}} \\
   &= \vel \approx \result{\velP{0}{2}}
\end{align}

\abc The kinetic energy is $e U = \SI{2.0}{\kilo\electronvolt}$, compared with the rest energy
\begin{align}
 \frac{E_\mathrm{kin}}{m_e c^2} &= \ratioF = \frac{\nce\times\U}{\ncme\times\left(\ncc\right)^2} \approx \pctP{0}{2}
\end{align}
This is only a small fraction: \emph{the classical formula is good enough}.
\end{abcliste}
