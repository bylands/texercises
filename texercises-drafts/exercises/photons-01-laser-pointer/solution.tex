\begin{abcliste}
\abc The energy of one photon is
\begin{align}
 E &= \frac{h c}{\lambda} = \EphF = \frac{\nch\times\ncc}{\la} = \Eph
\end{align}
The number of photons per second is the power divided by the energy of one photon:
\begin{align}
 \frac{N}{t} &= \frac{P}{E} = \NrF \\
   &= \frac{\Pl\times\la}{\nch\times\ncc} \\
   &= \Nr \approx \result{\NrP{0}{2}}
\end{align}

\abc In \tO:
\begin{align}
 N &= \NtF = \Nt \approx \result{\NtP{0}{2}}
\end{align}

\abc At \lbO\ each photon has less energy. The same power therefore needs $\result{\text{more photons}}$:
\begin{align}
 \frac{N}{t} &= \NbF = \frac{\Pl\times\lb}{\nch\times\ncc} = \Nb \approx \NbP{0}{2}
\end{align}
\end{abcliste}
