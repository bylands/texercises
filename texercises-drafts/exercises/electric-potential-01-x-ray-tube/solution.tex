\begin{abcliste}
\abc Through the voltage $U$, each electron (charge $e$) gains the energy
\begin{align}
 E_\mathrm{kin} &= \EkF = e\times\UO = \result{\EkP{3}{2}}
\end{align}

\abc The shortest wavelength belongs to a photon that carries the whole energy of an electron, $h c/\lambda = e U$:
\begin{align}
 \lambda_\mathrm{min} &= \lamF \\
   &= \frac{\nch\times\ncc}{\nce\times\U} \\
   &= \lam \approx \result{\lamP{-12}{2}}
\end{align}

\abc $\lambda_\mathrm{min}$ is inversely proportional to $U$: with a higher voltage, the shortest wavelength gets \emph{shorter}.
\end{abcliste}
