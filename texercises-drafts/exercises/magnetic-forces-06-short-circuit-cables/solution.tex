\begin{abcliste}
\abc The current flows out in one cable and back in the other: opposite currents $\result{\text{repel}}$ each other.

\abc The field of one long straight current at the other cable is
\begin{align}
 B &= \frac{\mu_0\cdot I}{2\pi\cdot d}
\end{align}
The force on the other cable per metre is $F/\ell = I\cdot B$:
\begin{align}
 \frac{F}{\ell} &= \FlF \\
   &= \frac{\ncmuo\times\left(\I\right)^2}{2\pi\times\d} \\
   &= \Fl \approx \result{\FlP{0}{2}}
\end{align}

\abc The force grows with $I^2$ (the field of one current acts on the other current):
\begin{align}
 k &= \kfF = \left(\frac{\I}{\Io}\right)^2 = \result{\kfP{0}{2}}
\end{align}
\end{abcliste}
