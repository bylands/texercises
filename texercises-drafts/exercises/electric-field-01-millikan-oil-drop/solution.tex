\begin{abcliste}
\abc The mass follows from the volume of a sphere:
\begin{align}
 m &= \mF \\
   &= \dens\times\frac{4}{3}\pi\times\left(\rd\right)^3 \\
   &= \m \approx \result{\mP{0}{2}}
\end{align}

\abc The drop hovers: the electric force $q E$, with $E = U/d$ between the plates, balances its weight,
\begin{align}
 q\,\frac{U}{d} = m g \quad\Rightarrow\quad q &= \qF \\
   &= \frac{\m\times\ncg\times\dp}{\U} \\
   &= \q \approx \result{\qP{0}{2}}
\end{align}

\abc
\begin{align}
 \frac{q}{e} &= \NqF = \frac{\q}{\nce} = \Nq \approx \result{\NqP{0}{1}}
\end{align}
The charge is a whole number of elementary charges.
\end{abcliste}
