\abc
\begin{align}
 \lambda &= \laF = \frac{\nch\times\ncc}{\EgO} = \la \approx \result{\laP{-12}{2}}
\end{align}

\abc The Compton shift with the Compton wavelength $\lambda_C = \frac{h}{m_e c}$ is
\begin{align}
 \Delta\lambda &= \dlF = \frac{\nch}{\ncme\times\ncc}\cdot\left(1-\cos\thtO\right) = \dl
\end{align}
The scattered photons therefore have the wavelength $\lambda' = \lambda+\Delta\lambda = \lb$ and the energy
\begin{align}
 E' &= \EsF = \frac{\nch\times\ncc}{\lb} = \Es \approx \result{\EsP{3}{3}}
\end{align}

\abc The electron gets the energy the photon lost:
\begin{align}
 \ssc{E}{kin} &= E-E' = \EgO - \EsP{3}{3} \approx \result{\EeP{3}{3}}
\end{align}
Scattered straight back, the photon gives the electron as much energy as it can (the Compton edge in a gamma spectrum).
\end{abcliste}
